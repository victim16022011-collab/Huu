local ActiveTween = nil
local TweenConnection = nil

local function StopCurrentTween()
    if ActiveTween then
        pcall(function()
            ActiveTween:Cancel()
        end)
        ActiveTween = nil
    end
    if TweenConnection then
        TweenConnection:Disconnect()
        TweenConnection = nil
    end
end

-- Hàm bay mượt không bị giật lùi vị trí
local function TweenTo(targetCFrame)
    pcall(function()
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local distance = (hrp.Position - targetCFrame.Position).Magnitude
        if distance <= 15 then
            hrp.CFrame = targetCFrame
            return
        end

        -- Nếu đang có tween chạy hoặc bị gián đoạn, hủy cái cũ để tạo luồng mới chính xác
        if ActiveTween then
            return 
        end

        local speed = getgenv().DragonHub.TweenSpeed or 300
        local tweenTime = distance / speed
        local tweenInfo = TweenInfo.new(tweenTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)

        ActiveTween = TweenService:Create(hrp, tweenInfo, { CFrame = targetCFrame })
        
        -- Khóa vận tốc chống server anti-cheat kéo giật về chỗ cũ
        TweenConnection = RunService.Heartbeat:Connect(function()
            if hrp then
                hrp.Velocity = Vector3.new(0, 0, 0)
            end
        end)

        ActiveTween:Play()
        ActiveTween.Completed:Connect(function()
            if TweenConnection then
                TweenConnection:Disconnect()
                TweenConnection = nil
            end
            ActiveTween = nil
        end)
    end)
end

-- Hàm cầm vũ khí
local function EquipSelectedWeapon()
    pcall(function()
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        if not char or not hum or not backpack then return end

        local desiredType = getgenv().DragonHub.WeaponType
        local currentTool = char:FindFirstChildOfClass("Tool")
        if currentTool then
            local wType = currentTool:GetAttribute("WeaponType") or currentTool.ToolTip
            if wType == desiredType then return end
        end

        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                local wType = item:GetAttribute("WeaponType") or item.ToolTip
                if wType == desiredType then
                    hum:EquipTool(item)
                    break
                end
            end
        end
    end)
end

end

local function PerformFastAttack()
    pcall(function()
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local enemiesFolder = Workspace:FindFirstChild("Enemies")
        local primaryTargetPart = nil
        local hitTable = {}

        if enemiesFolder then
            for _, mob in ipairs(enemiesFolder:GetChildren()) do
                local mobHrp = mob:FindFirstChild("HumanoidRootPart")
                local mobHum = mob:FindFirstChildOfClass("Humanoid")
                if mobHrp and mobHum and mobHum.Health > 0 then
                    if (mobHrp.Position - hrp.Position).Magnitude <= 65 then
                        if not primaryTargetPart then
                            primaryTargetPart = mobHrp
                        else
                            table.insert(hitTable, { mob, mobHrp })
                        end
                    end
                end
            end
        end

        if primaryTargetPart then
            if RegisterAttack and RegisterHit then
                RegisterAttack:FireServer(0)
                RegisterHit:FireServer(primaryTargetPart, hitTable)
            end
            local heldTool = char:FindFirstChildOfClass("Tool")
            if heldTool then
                local leftClick = heldTool:FindFirstChild("LeftClickRemote")
                if leftClick and leftClick:IsA("RemoteEvent") then
                    leftClick:FireServer(Vector3.new(0.01, -500, 0.01), 1, true)
                    leftClick:FireServer(false)
                end
            end
        end
    end)
end

local function BringNearbyMobs(targetName, targetCFrame)
    if not getgenv().DragonHub.BringMob then return end
    pcall(function()
        if sethiddenproperty then
            sethiddenproperty(LocalPlayer, "SimulationRadius", math.huge)
        end
        local enemiesFolder = Workspace:FindFirstChild("Enemies")
        if not enemiesFolder then return end

        for _, mob in ipairs(enemiesFolder:GetChildren()) do
            if mob.Name == targetName then
                local mobHrp = mob:FindFirstChild("HumanoidRootPart")
                local mobHum = mob:FindFirstChildOfClass("Humanoid")
                if mobHrp and mobHum and mobHum.Health > 0 then
                    if (mobHrp.Position - targetCFrame.Position).Magnitude <= 300 then
                        mobHrp.CFrame = targetCFrame
                        mobHrp.CanCollide = false
                        mobHum.WalkSpeed = 0
                        mobHum.JumpPower = 0
                    end
                end
            end
        end
    end)
end

local function EnableDragging(handle, window)
    local dragging, dragInput, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = window.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            window.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

-- 🧠 LOGIC AUTO FARM LEVEL CHUẨN SEA 3 (TỰ ĐỌC LEVEL & NHẬN QUEST)
task.spawn(function()
    while task.wait(0.3) do
        pcall(function()
            if not getgenv().DragonHub.AutoFarmLevel then return end

            local myLevel = LocalPlayer.Data.Level.Value
            local questData = nil

            -- Tự động phân loại quest dựa theo Level hiện tại của người chơi ở Sea 3
            if myLevel >= 2550 then
                questData = { Name = "TikiQuest3", Level = 2, Mon = "Skull Slayer", QuestPos = CFrame.new(-16661.89, 105.28, 1576.69), MonPos = CFrame.new(-16885.20, 114.12, 1627.94) }
            elseif myLevel >= 2525 then
                questData = { Name = "TikiQuest2", Level = 2, Mon = "Isle Champion", QuestPos = CFrame.new(-16539.07, 55.68, 1051.57), MonPos = CFrame.new(-16641.67, 235.78, 1031.28) }
            elseif myLevel >= 2450 then
                questData = { Name = "TikiQuest1", Level = 1, Mon = "Isle Outlaw", QuestPos = CFrame.new(-16547.74, 61.13, -173.41), MonPos = CFrame.new(-16442.81, 116.13, -264.46) }
            elseif myLevel >= 2300 then
                questData = { Name = "ChocQuest1", Level = 1, Mon = "Cocoa Warrior", QuestPos = CFrame.new(233.22, 29.87, -12201.23), MonPos = CFrame.new(-21.55, 80.57, -12352.38) }
            else
                -- Mặc định bãi quái cấp cao an toàn nếu level thấp hơn
                questData = { Name = "PiratePortQuest", Level = 1, Mon = "Pirate Millionaire", QuestPos = CFrame.new(-290.07, 42.90, 5581.58), MonPos = CFrame.new(-245.99, 47.30, 5584.10) }
            end

            if questData and CommF_ then
                local questGui = LocalPlayer.PlayerGui:FindFirstChild("Main") and LocalPlayer.PlayerGui.Main:FindFirstChild("Quest")
                local hasActiveQuest = questGui and questGui.Visible

                if not hasActiveQuest then
                    if (LocalPlayer.Character.HumanoidRootPart.Position - questData.QuestPos.Position).Magnitude > 10 then
                        TweenTo(questData.QuestPos)
                    else
                        CommF_:InvokeServer("AbandonQuest")
                        CommF_:InvokeServer("StartQuest", questData.Name, questData.Level)
                        task.wait(0.5)
                    end
                else
                    -- Đang có quest -> Tìm quái tương ứng
                    CheckAndEnableBuso()
                    EquipSelectedWeapon()

                    local enemies = Workspace:FindFirstChild("Enemies")
                    local targetMob = nil
                    if enemies then
                        for _, m in ipairs(enemies:GetChildren()) do
                            if m.Name == questData.Mon and m:FindFirstChild("HumanoidRootPart") and m:FindFirstChildOfClass("Humanoid") and m.Humanoid.Health > 0 then
                                targetMob = m
                                break
                            end
                        end
                    end

                    if targetMob then
                        TweenTo(targetMob.HumanoidRootPart.CFrame * CFrame.new(0, getgenv().DragonHub.FarmDistance, 0))
                        BringNearbyMobs(questData.Mon, targetMob.HumanoidRootPart.CFrame)
                    else
                        TweenTo(questData.MonPos * CFrame.new(0, 15, 0))
                    end
                end
            end
        end)
    end
end)

-- Luồng Auto Farm Quái Chỉ Định & Boss
task.spawn(function()
    while task.wait(0.2) do
        pcall(function()
            if getgenv().DragonHub.AutoFarmLevel then return end -- Ưu tiên Level farm trước

            if getgenv().DragonHub.AutoFarmMob then
                local mobName = getgenv().DragonHub.SelectedMob
                local mobData = MobDatabase[mobName]
                local enemies = Workspace:FindFirstChild("Enemies")
                local foundTarget = nil

                if enemies then
                    for _, m in ipairs(enemies:GetChildren()) do
                        if m.Name == mobName and m:FindFirstChild("HumanoidRootPart") and m:FindFirstChildOfClass("Humanoid") and m.Humanoid.Health > 0 then
                            foundTarget = m
                            break
                        end
                    end
                end

                if foundTarget then
                    CheckAndEnableBuso()
                    EquipSelectedWeapon()
                    TweenTo(foundTarget.HumanoidRootPart.CFrame * CFrame.new(0, getgenv().DragonHub.FarmDistance, 0))
                    BringNearbyMobs(mobName, foundTarget.HumanoidRootPart.CFrame)
                elseif mobData then
                    TweenTo(mobData.Pos * CFrame.new(0, 25, 0))
                end

            elseif getgenv().DragonHub.AutoFarmBoss then
                local bossName = getgenv().DragonHub.SelectedBoss
                local bossDefaultPos = BossDatabase[bossName]
                local enemies = Workspace:FindFirstChild("Enemies")
                local foundBoss = nil

                if enemies then
                    for _, b in ipairs(enemies:GetChildren()) do
                        if string.find(b.Name, bossName) and b:FindFirstChild("HumanoidRootPart") and b:FindFirstChildOfClass("Humanoid") and b.Humanoid.Health > 0 then
                            foundBoss = b
                            break
                        end
                    end
                end

                if foundBoss then
                    CheckAndEnableBuso()
                    EquipSelectedWeapon()
                    TweenTo(foundBoss.HumanoidRootPart.CFrame * CFrame.new(0, getgenv().DragonHub.FarmDistance, 0))
                elseif bossDefaultPos then
                    TweenTo(bossDefaultPos * CFrame.new(0, 25, 0))
                end
            end
        end)
    end
end)

-- Luồng Sự Kiện, Thức Tỉnh, Trái Cây & ESP
task.spawn(function()
    while task.wait(0.8) do
        pcall(function()
            local cfg = getgenv().DragonHub
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")

            if cfg.AutoRaceV3 and CommE then CommE:FireServer("ActivateAbility") end
            if cfg.AutoAwakening then
                local backpack = LocalPlayer:FindFirstChild("Backpack")
                local awk = (backpack and backpack:FindFirstChild("Awakening")) or (char and char:FindFirstChild("Awakening"))
                if awk and awk:FindFirstChild("RemoteEvent") then awk.RemoteEvent:FireServer() end
            end
            if cfg.AutoDragonHunter and DragonHunterRF then DragonHunterRF:InvokeServer({ Context = "RequestQuest" }) end
            if cfg.AutoDragonEmber then
                if DragonDojoEmber then DragonDojoEmber:FireServer() end
                if CollectedDragonEgg then CollectedDragonEgg:FireServer() end
            end
            if cfg.AutoKitsunePray then
                if TouchKitsuneStatue then TouchKitsuneStatue:FireServer() end
                if KitsuneStatuePrayRF then KitsuneStatuePrayRF:InvokeServer() end
                if CollectBlueEmber then CollectBlueEmber:FireServer() end
            end
            if cfg.AutoCollectBerries and ClaimBerryRF then ClaimBerryRF:InvokeServer() end

            if cfg.AutoGrabFruits and hrp then
                for _, item in ipairs(Workspace:GetChildren()) do
                    if item:IsA("Tool") and item:FindFirstChild("Handle") then item.Handle.CFrame = hrp.CFrame end
                end
            end

            if cfg.AutoFirePrompts and hrp and fireproximityprompt then
                for _, obj in ipairs(Workspace:GetDescendants()) do
                    if obj:IsA("ProximityPrompt") and obj.Enabled then
                        local parentPart = obj.Parent
                        if parentPart and parentPart:IsA("BasePart") then
                            if (parentPart.Position - hrp.Position).Magnitude <= obj.MaxActivationDistance + 5 then
                                fireproximityprompt(obj)
                            end
                        end
                    end
                end
            end

            if CommF_ then
                if cfg.AutoStatsMelee then CommF_:InvokeServer("AddPoint", "Melee", 3) end
                if cfg.AutoStatsDefense then CommF_:InvokeServer("AddPoint", "Defense", 3) end
                if cfg.AutoStatsSword then CommF_:InvokeServer("AddPoint", "Sword", 3) end
                if cfg.AutoStatsGun then CommF_:InvokeServer("AddPoint", "Gun", 3) end
                if cfg.AutoStatsFruit then CommF_:InvokeServer("AddPoint", "Demon Fruit", 3) end
            end
        end)
    end
end)
